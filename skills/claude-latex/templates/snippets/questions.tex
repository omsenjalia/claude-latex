\section*{Q-2 Do as directed: [A]}
\begin{enumerate}[label=\arabic*.]
  \item Expand $y=\log x$ in powers of $(x-1)$.
  \item Expand
        \[ f(x)=x^{4}-11x^{3}+7x^{2}-19x+14 \]
        in powers of $(x-3)$.
  \item Expand
        \[ \sin\left(\frac{\pi}{4}+x\right) \]
        in powers of $x$. Hence find the values of
        \[ \sin 44^{\circ}\quad\text{and}\quad\sin 46^{\circ}. \]
\end{enumerate}
